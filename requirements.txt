
from num2words import num2words
import pandas as pd
from reportlab.lib.pagesizes import letter
from reportlab.lib import colors
from reportlab.lib.styles import getSampleStyleSheet, ParagraphStyle
from reportlab.platypus import SimpleDocTemplate, Paragraph, Spacer, Table, TableStyle, Image
import streamlit as st
import os

st.title("RAJESH Invoice Generator")

# User Inputs for Invoice Header
invoice_no = st.text_input("Invoice No", "001")
invoice_date = st.date_input("Date")
customer_name = st.text_input("Customer Name (ISSUED TO)", "Enter Customer Name & Address")

# Dynamic Items Table input using Session State
if 'items' not in st.session_state:
    st.session_state.items = [{'description': '', 'price': 0.0, 'quantity': 1}]

def add_item():
    st.session_state.items.append({'description': '', 'price': 0.0, 'quantity': 1})

def remove_item(index):
    if len(st.session_state.items) > 1:
        st.session_state.items.pop(index)

st.subheader("Description / Items")
for i, item in enumerate(st.session_state.items):
    cols = st.columns([3, 1, 1, 1])
    with cols[0]:
        st.session_state.items[i]['description'] = st.text_input(f"Description {i+1}", value=item['description'], key=f"desc_{i}")
    with cols[1]:
        st.session_state.items[i]['price'] = st.number_input(f"Price (INR) {i+1}", min_value=0.0, value=float(item['price']), key=f"price_{i}")
    with cols[2]:
        st.session_state.items[i]['quantity'] = st.number_input(f"Quantity {i+1}", min_value=1, value=int(item['quantity']), key=f"qty_{i}")
    with cols[3]:
        if st.button("Delete", key=f"del_{i}"):
            remove_item(i)
            st.rerun()

if st.button("Add Item"):
    add_item()
    st.rerun()

def generate_pdf(filename, inv_no, inv_date, cust_name, items_list):
    doc = SimpleDocTemplate(filename, pagesize=letter, rightMargin=30, leftMargin=30, topMargin=30, bottomMargin=30)
    story = []
    styles = getSampleStyleSheet()

    # Header Table: Logo on Left (proportional width/height), INVOICE Title on Right
    logo_path = "logo.png" # Make sure to save your uploaded logo with this name
    if os.path.exists(logo_path):
        # Adjusted size to maintain the logo's aspect ratio cleanly
        logo_img = Image(logo_path, width=70, height=70)
    else:
        logo_img = Paragraph("<b>RAJESH LOGO</b>", styles['Normal'])

    invoice_title_style = ParagraphStyle(
        'InvoiceTitle',
        parent=styles['Heading1'],
        fontSize=26,
        alignment=2, # Right align
        textColor=colors.HexColor("#222222")
    )
    invoice_header_text = Paragraph("<b>INVOICE</b>", invoice_title_style)

    header_table = Table([[logo_img, invoice_header_text]], colWidths=[250, 288])
    header_table.setStyle(TableStyle([
        ('VALIGN', (0,0), (-1,-1), 'MIDDLE'),
        ('ALIGN', (1,0), (1,0), 'RIGHT'),
    ]))
    story.append(header_table)
    story.append(Spacer(1, 15))

    # Issued To & Details Table
    issued_text = f"<b>ISSUED TO:</b><br/>{cust_name.replace(chr(10), '<br/>')}"
    meta_text = f"<b>Invoice No:</b> {inv_no}<br/><b>Date:</b> {inv_date.strftime('%d-%m-%Y')}"

    info_table = Table([[Paragraph(issued_text, styles['Normal']), Paragraph(meta_text, styles['Normal'])]], colWidths=[300, 238])
    info_table.setStyle(TableStyle([
        ('VALIGN', (0,0), (-1,-1), 'TOP'),
        ('BACKGROUND', (0,0), (-1,-1), colors.HexColor("#FAFAFA")),
        ('PADDING', (0,0), (-1,-1), 8),
        ('BOX', (0,0), (-1,-1), 0.5, colors.HexColor("#CCCCCC")),
    ]))
    story.append(info_table)
    story.append(Spacer(1, 20))

    # Items Table Header & Rows
    table_data = [["DESCRIPTION", "PRICE (INR)", "QUANTITY", "AMOUNT"]]
    
    total_amount = 0.0
    for item in items_list:
        amount = item['price'] * item['quantity']
        total_amount += amount
        table_data.append([
            item['description'],
            f"₹ {item['price']:,.2f}",
            str(item['quantity']),
            f"₹ {amount:,.2f}"
        ])

    items_table = Table(table_data, colWidths=[228, 100, 80, 130])
    items_table.setStyle(TableStyle([
        ('BACKGROUND', (0,0), (-1,0), colors.HexColor("#222222")),
        ('TEXTCOLOR', (0,0), (-1,0), colors.whitesmoke),
        ('ALIGN', (0,0), (-1,-1), 'CENTER'),
        ('ALIGN', (0,1), (0,-1), 'LEFT'),
        ('FONTNAME', (0,0), (-1,0), 'Helvetica-Bold'),
        ('BOTTOMPADDING', (0,0), (-1,0), 8),
        ('GRID', (0,0), (-1,-1), 0.5, colors.HexColor("#DDDDDD")),
        ('VALIGN', (0,0), (-1,-1), 'MIDDLE'),
    ]))
    story.append(items_table)
    story.append(Spacer(1, 15))

    # Total & Rupee in Word Table
    try:
        rupees_in_words = num2words(int(total_amount), lang='en_IN').capitalize() + " Rupees Only"
    except:
        rupees_in_words = ""

    summary_data = [
        ["TOTAL:", f"₹ {total_amount:,.2f}"],
        ["RUPEE IN WORD:", rupees_in_words]
    ]

    summary_table = Table(summary_data, colWidths=[120, 418])
    summary_table.setStyle(TableStyle([
        ('FONTNAME', (0,0), (-1,-1), 'Helvetica-Bold'),
        ('ALIGN', (0,0), (0,-1), 'RIGHT'),
        ('ALIGN', (1,0), (1,-1), 'LEFT'),
        ('BACKGROUND', (0,0), (-1,-1), colors.HexColor("#F1F1F1")),
        ('BOX', (0,0), (-1,-1), 0.5, colors.HexColor("#CCCCCC")),
        ('PADDING', (0,0), (-1,-1), 6),
    ]))
    story.append(summary_table)

    doc.build(story)

if st.button("Generate Invoice PDF"):
    pdf_filename = "rajesh_invoice.pdf"
    generate_pdf(pdf_filename, invoice_no, invoice_date, customer_name, st.session_state.items)
    
    with open(pdf_filename, "rb") as pdf_file:
        st.download_button(
            label="Download Invoice PDF",
            data=pdf_file,
            file_name=pdf_filename,
            mime="application/pdf"
        )
    st.success("Invoice generated successfully with your RK Logo!")
